MCQ-র প্রিয় ★
সংজ্ঞা হেডার ফাইল (Header File) (.h) একাধিক সোর্স ফাইলের ভাগ করা ডিক্লারেশন রাখে: ফাংশন প্রোটোটাইপ, ম্যাক্রো, কনস্ট্যান্ট ও টাইপ সংজ্ঞা (typedef, struct)। #include লাইনটির জায়গায় ফাইলের পুরো টেক্সট বসিয়ে দেয় — কম্পাইলের আগেই। হেডারে কোনো কম্পাইলড কোড থাকে না: printf, sqrt … -এর বডি থাকে লাইব্রেরিতে, যা জোড়ে লিংকার (Linker)।
হেডারকী দেয় · যে নামগুলো চিনতে হবে
<stdio.h>স্ট্যান্ডার্ড I/O: printf, scanf, puts, fgets, getchar, fopen, FILE, EOF
<conio.h>কনসোল I/O, স্ট্যান্ডার্ড C নয় (Turbo C/DOS, MSVC): clrscr, getch, getche
<math.h>double-এ গণিত: sqrt, pow, fabs, ceil, floor, sin, log (gcc: -lm দিয়ে লিংক)
<string.h>স্ট্রিং ও মেমোরি ব্লক: strlen, strcpy, strcat, strcmp, strstr, memcpy
<stdlib.h>সাধারণ ইউটিলিটি: malloc, calloc, free, atoi, rand, exit, abs, qsort
<ctype.h>ক্যারেক্টার পরীক্ষা ও কেস: isalpha, isdigit, isspace, toupper, tolower
<time.h>তারিখ ও সময়: time, clock, difftime, CLOCKS_PER_SEC
<limits.h>ইন্টিজারের সীমা: INT_MAX, INT_MIN, CHAR_BIT, LONG_MAX
<float.h>ফ্লোটিং-পয়েন্টের সীমা: FLT_MAX, DBL_MAX, FLT_EPSILON, FLT_DIG
<stddef.h> · <stdbool.h>NULL, size_t, ptrdiff_t · bool, true, false (C99)
/* myheader.h */ #ifndef MYHEADER_H /* include guard */ #define MYHEADER_H int add(int a, int b); /* prototype */ #define MAX 100 #endif /* main.c */ #include <stdio.h> /* system dirs */ #include "myheader.h" /* own dir 1st */
<…> বনাম "…" <stdio.h> খোঁজা হয় কেবল স্ট্যান্ডার্ড (সিস্টেম) ইনক্লুড ডিরেক্টরিতে — লাইব্রেরি হেডারের জন্য। "my.h" খোঁজা হয় আগে নিজের ফোল্ডারে, তারপর সিস্টেম ডিরেক্টরিতে — নিজের হেডারের জন্য। ইনক্লুড গার্ড (Include Guard) দ্বিতীয় কপি আটকায়, তাই "redefinition" এরর হয় না; #pragma once একই কাজ করে (স্ট্যান্ডার্ড নয়, তবে প্রায় সব কম্পাইলারে চলে)।
মনে রাখার কৌশল "অ্যাঙ্গেল ব্র্যাকেট = লাইব্রেরির তাক; কোটেশন = নিজের টেবিল, আগে টেবিল দেখো, তারপর তাক।"
পরীক্ষার ফাঁদ হেডার তার লাইব্রেরি লিংক করে না: <math.h> থাকলেও gcc-তে -lm লাগে, নইলে "undefined reference"। conio.h, clrscr(), getch() ISO C নয়।
সমাধানকৃত উদাহরণ ✍ কোন হেডার? INT_MAX → limits.h · toupper → ctype.h · malloc → stdlib.h · fabs → math.h কিন্তু abs → stdlib.h।

২প্রিপ্রসেসর ডিরেক্টিভ ও ম্যাক্রো

আউটপুট প্রশ্ন ★
সংজ্ঞা প্রিপ্রসেসর (Preprocessor) চলে কম্পাইলারের আগে এবং # দিয়ে শুরু হওয়া প্রতিটি লাইন (ডিরেক্টিভ, Directive) সামলায়। এর কাজ পুরোপুরি টেক্সটের — ফাইল যুক্ত করা, ম্যাক্রো (Macro) প্রতিস্থাপন, শর্তসাপেক্ষ কম্পাইলেশন — কমেন্ট সরানোর পরে। ডিরেক্টিভ লাইনের শেষে শেষ হয়: সেমিকোলন নেই (লম্বা হলে \ দিয়ে পরের লাইনে)।
ডিরেক্টিভকাজ
#includeফাইলের টেক্সট যুক্ত করে
#defineকনস্ট্যান্ট বা ম্যাক্রো: #define PI 3.1416
#undefম্যাক্রোর সংজ্ঞা মুছে দেয়
#ifdef · #ifndefসংজ্ঞায়িত হলে · না হলে কম্পাইল
#if #elif #else #endifধ্রুবক ইন্টিজার শর্তে কম্পাইল; defined(X) চলে
#error msgবার্তা দিয়ে কম্পাইল থামায়
#pragma · #lineকম্পাইলার-নির্দিষ্ট নির্দেশ (once, pack) · লাইন নম্বর বদল
#define DEBUG 1 #define VERSION 2 #ifdef DEBUG printf("Debug mode ON\n"); #endif #if VERSION >= 2 newApi(); #elif VERSION == 1 oldApi(); #else #error "unknown VERSION" #endif #undef DEBUG /* gone from here */

পূর্বনির্ধারিত ম্যাক্রো (Predefined Macro): __FILE__ ফাইলের নাম (স্ট্রিং) · __LINE__ লাইন নম্বর (int) · __DATE__ "Oct 14 2026" · __TIME__ "hh:mm:ss" · __STDC__ মানসম্মত কম্পাইলারে 1 · __STDC_VERSION__ 199901L (C99), 201112L (C11)।

বিষয়ম্যাক্রোফাংশন
কখনপ্রিপ্রসেসিং: টেক্সটরান টাইম: কল
টাইপ চেকনেইআছে (প্রোটোটাইপ)
আর্গুমেন্টপ্রতি ব্যবহারে আবার মূল্যায়নএকবার মূল্যায়ন
গতি · আকারকলের খরচ নেই · কোড বাড়েকলের খরচ · এক কপি
অ্যাড্রেসনেইফাংশন পয়েন্টার
# ও ## # স্ট্রিং বানায়: #define STR(x) #x, STR(Dhaka) → "Dhaka"। ## টোকেন জোড়ে: #define CAT(a,b) a##b, CAT(var,1) → var1। ফাংশন-সদৃশ ম্যাক্রোর চেয়ে নিরাপদ: inline ফাংশন (C99)।
মনে রাখার কৌশল "ভেতরে ব্র্যাকেট, বাইরেও ব্র্যাকেট": প্রতিটি প্যারামিটার ও পুরো বডি মুড়ে দাও — ((x)*(x))।
পরীক্ষার ফাঁদ #define PI 3.14; দিলে ;-ও বসে যায়। SQUARE(i++) i-কে দুবার বাড়ায়: অনির্ধারিত। #define SQ (x)-এ ফাঁকা থাকলে তা অবজেক্ট-সদৃশ।
সমাধানকৃত উদাহরণ ✍ আউটপুট #define SQ(x) (x)*(x): 100/SQ(5) → 100/(5)*(5) = 20×5 = 100, 4 নয়। পুরো ব্র্যাকেটে: 4।

৩পয়েন্টার: &, *, প্রকার ও const

সবচেয়ে বেশি আসে ★
সংজ্ঞা পয়েন্টার (Pointer) এমন একটি ভেরিয়েবল যা অন্য অবজেক্টের মেমোরি অ্যাড্রেস রাখে: type *name;। & (অ্যাড্রেস-অফ) অ্যাড্রেস দেয়; * (ইনডিরেকশন / ডিরেফারেন্স, "ওই অ্যাড্রেসের মান") সেই অ্যাড্রেসের অবজেক্ট দেয়। পয়েন্টারের নিজের আকার মেশিন ঠিক করে — 32-বিটে 4 বাইট, 64-বিটে 8 — যেটাকেই দেখাক; তার টাইপ ঠিক করে * কত বাইট পড়বে আর +1 কতদূর সরবে।
int a = 10; int *p; /* declare */ p = &a; /* address of a */ printf("%d", *p); /* 10 */ *p = 20; /* a is now 20 */ int **q = &p; /* ptr to ptr */ printf("%d", **q);/* 20 */ int *x, y; /* x ptr, y int! */
এক্সপ্রেশনমান

p · *p · &p1000 · 20 · 2000
*&a (পরস্পর কাটে)20
q · *q · **q2000 · 1000 · 20
printf("%p", (void *)p)1000 ছাপে
প্রকারকীবিপদ ও প্রতিকার
NULLint *p = NULL; কোথাও দেখায় নাডিরেফারেন্স → ক্র্যাশ; আগে p != NULL দেখো
void (জেনেরিক)যেকোনো অবজেক্টের অ্যাড্রেস রাখে; malloc এটাই দেয়কাস্ট ছাড়া *v বা v+1 নয়: *(int *)v
ওয়াইল্ড (Wild)কখনো ইনিশিয়ালাইজ হয়নি: আবর্জনা অ্যাড্রেসNULL বা বৈধ অ্যাড্রেস দাও
ড্যাংলিং (Dangling)অবজেক্ট আর নেই: free(p)-এর পরে, বা লোকালের অ্যাড্রেস ফেরত দিলেfree-এর পরে p = NULL;
ডিক্লারেশন*p=5p++
const int *p = int const *p✗✓
int *const p = &a;✓✗
const int *const p✗✗
দেখতে একরকম int *p[3]: ৩টি পয়েন্টারের অ্যারে। int (*p)[3]: ৩টি int-এর অ্যারের দিকে একটি পয়েন্টার। int (*fp)(int): ফাংশন পয়েন্টার। ফাংশনে &x পাঠানো: আগের লেসন।
মনে রাখার কৌশল "& জিজ্ঞেস করে কোথায়; * জিজ্ঞেস করে সেখানে কী।" ডান থেকে বাঁয়ে পড়ো: int *const p = "p হলো int-এর const পয়েন্টার"।
পরীক্ষার ফাঁদ int *x, y;-এ কেবল x পয়েন্টার: * নামের সঙ্গে লাগে। NULL বা ওয়াইল্ড ডিরেফারেন্স কম্পাইল হয়, রান টাইমে ভাঙে।
সমাধানকৃত উদাহরণ ✍ ট্রেস int a = 5, *p = &a, **q = &p; **q += *p; → দুটোই a, তাই a = 5 + 5 = 10।

সংখ্যাভিত্তিক MCQ ★
সংজ্ঞা পয়েন্টারে ইন্টিজার n যোগ করলে তা n টি উপাদান সরে, n বাইট নয়: address(p + n) = address(p) + n × sizeof(*p)। একই অ্যারের দুটি পয়েন্টার বিয়োগ করলে মাঝের উপাদানের সংখ্যা পাওয়া যায় (টাইপ ptrdiff_t)। ফল নির্ধারিত কেবল একটি অ্যারের ভেতরে, আর শেষের ঠিক পরের অবস্থানে।

char *11001
short *21002

double *81008
int ** (64-বিট)81008

সংজ্ঞা int m[R][C] হলো R টি সারির অ্যারে, প্রতিটি সারি C টি int-এর অ্যারে। মেমোরি একটি সরল রেখা, আর C উপাদানগুলো রাখে রো-মেজর ক্রমে (Row-major Order): আগে পুরো সারি ০, তারপর সারি ১, … কেবল প্রথম (সারির) আকার বাদ দেওয়া যায় — int m[][3] = {…};, void f(int m[][3], int rows) — কারণ প্রতিটি সারি কোথায় শুরু তা বের করতে কলামের আকার লাগবেই।

লিখিত প্রশ্ন C-তে অ্যারে ও পয়েন্টারের সম্পর্ক ব্যাখ্যা করো। arr[i] কীভাবে মূল্যায়িত হয় দেখাও, অ্যারে ও পয়েন্টারে sizeof কেন ভিন্ন ফল দেয় বলো, A[i][j]-এর রো-মেজর অ্যাড্রেস নির্ণয় করো, এবং পয়েন্টার দিয়ে স্ট্রিংয়ের দৈর্ঘ্য ফেরত দেয় এমন একটি ফাংশন লেখো। ১০ নম্বর
  1. রূপান্তর (২) — এক্সপ্রেশনে অ্যারের নাম হয়ে যায় &arr[0]; arr[i]-এর সংজ্ঞাই *(arr + i), আর arr + i যোগ করে i × sizeof(int) বাইট। ২
  2. sizeof (২) — sizeof/&-এর সঙ্গে রূপান্তর হয় না: int arr[5] → 20; পয়েন্টার (বা অ্যারে প্যারামিটার) → 64-বিটে 8; নামটিতে অ্যাসাইন বা ইনক্রিমেন্ট চলে না। ২
  3. অ্যাড্রেস (৩) — সারিগুলো পরপর থাকে: N উপাদানের i টি পুরো সারি পেরিয়ে তারপর j: LOC = B + W[N(i − Lr) + (j − Lc)]; যেমন int A[5][4] 1000-এ → A[3][2] = 1056। ৩
  4. ফাংশন (৩) — size_t len(const char *s){ const char *p = s; while (*p) p++; return p - s; }: '\0' পর্যন্ত এগোও, পয়েন্টারের পার্থক্যই দৈর্ঘ্য। ৩

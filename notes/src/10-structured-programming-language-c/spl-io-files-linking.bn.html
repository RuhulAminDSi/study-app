মূল বিষয় ★
সংজ্ঞা স্ট্রিম (stream) হলো বাইটের যেকোনো উৎস বা গন্তব্য — কীবোর্ড, স্ক্রিন, ফাইল, পাইপ — দেখার C-র একক উপায়, যা একটি FILE * দিয়ে ধরা হয়। প্রতিটি হোস্টেড C প্রোগ্রাম শুরু হয় আগে থেকেই খোলা তিনটি স্ট্রিম নিয়ে, যা <stdio.h>-এ ঘোষিত: stdin (কীবোর্ড), stdout (স্ক্রিন; টার্মিনালে লাইন-বাফারড, ফাইলে পুরো বাফারড) এবং stderr (স্ক্রিন, আনবাফারড — সঙ্গে সঙ্গে দেখায়)।
পরীক্ষার ফাঁদ (১) stderr কোনো লগ ফাইল নয়: এটিও স্ক্রিনে যায়, কিন্তু prog > out.txt কেবল stdout সরায়, তাই এরর তখনো দেখা যায়। (২) stdout-এর বাফারে থাকা লেখা প্রোগ্রাম ক্র্যাশ করলে হারিয়ে যায় — fflush(stdout) জোর করে বের করে দেয়।

ফাঁদে ভরা ★
সংজ্ঞা scanf stdin থেকে অক্ষর পড়ে, ফরম্যাট অনুযায়ী রূপান্তর করে এবং প্রতিটি ফল একটি পয়েন্টারের মাধ্যমে জমা রাখে — তাই অ্যারে নয় এমন প্রতিটি আর্গুমেন্টে & লাগে। এটি ফেরত দেয় যতগুলো আইটেম বসানো হলো সেই সংখ্যা, অথবা প্রথম রূপান্তরের আগেই ইনপুট ব্যর্থ হলে EOF। %d, %f ও %s শুরুর ফাঁকা জায়গা (white space) বাদ দেয়; %c দেয় না।

সমাধানকৃত উদাহরণ ✍ scanf কী ফেরত দেয়? int a = 0, b = 0, r = scanf("%d %d", &a, &b); (ক) ইনপুট 10 20 → r = 2, a = 10, b = 20। (খ) ইনপুট 10 abc → r = 1, b থাকে 0 এবং "abc" পরের পড়ার জন্য বাফারেই থেকে যায়। (গ) ইনপুট abc → r = 0। (ঘ) ইনপুট শেষ (Ctrl+Z / Ctrl+D) → r = EOF। নিরাপদ লুপ: while (scanf("%d", &x) == 1) sum += x;

৩ফাইল অ্যাক্সেস: FILE *, fopen মোড, টেক্সট বনাম বাইনারি

সবচেয়ে বেশি আসে ★
সংজ্ঞা ফাইল ব্যবহার করা হয় একটি FILE স্ট্রাকচারের পয়েন্টার দিয়ে (<stdio.h>-এ ঘোষিত), যেটি রাখে বাফার, বর্তমান অবস্থান (position) এবং ফাইল-শেষ ও এরর ফ্ল্যাগ। প্রতিটি ফাইলের কাজের তিন ধাপ: fopen(name, mode) দিয়ে খোলা, যা ব্যর্থ হলে NULL ফেরত দেয়; প্রসেস করা; fclose(fp) দিয়ে বন্ধ, যা বাফার খালি করে 0 ফেরত দেয় (ব্যর্থ হলে EOF)।
মোডফাইল থাকলেনা থাকলেঅনুমতি
"r"শুরু থেকে খোলেNULLপড়া
"w"শূন্যে কেটে ফেলেতৈরি করেলেখা
"a"রাখে; প্রতিটি লেখা যায় শেষেতৈরি করেযোগ (append)
"r+"খোলে, কাটে নাNULLপড়া + লেখা
"w+"কেটে ফেলেতৈরি করেলেখা + পড়া
"a+"রাখে; যেকোনো জায়গা পড়ে, শেষে লেখেতৈরি করেপড়া + যোগ

বাইনারি-র জন্য b যোগ করো: "rb", "wb", "ab", "r+b" (= "rb+"), "w+b", "a+b"। C11 যোগ করেছে x ("wx"): ওভাররাইট না করে ব্যর্থ হয়। মনে রাখো printf(…) ≡ fprintf(stdout, …)।

টেক্সট বনাম বাইনারি টেক্সট মোডে লাইব্রেরি লাইনের শেষ বদলাতে পারে — Windows-এ '\n' লেখা হয় "\r\n" হিসেবে আর পড়ার সময় আবার '\n'। বাইনারি মোড বাইট হুবহু সরায়। Linux/UNIX-এ দুই মোড একই। fprintf(fp, "%d", 12345) লেখে 5টি অক্ষর; fwrite(&n, sizeof n, 1, fp) লেখে sizeof(int) বাইট (বেশিরভাগ সিস্টেমে 4)।
পরীক্ষার ফাঁদ (১) "w" পুরোনো লেখা মুছে দেয়; রাখতে চাইলে "a"। (২) "r+" কখনো ফাইল তৈরি করে না। (৩) fopen ব্যর্থতা জানায় NULL দিয়ে, EOF বা −1 নয়। (৪) fclose ভুলে গেলে বাফারের ডেটা হারাতে পারে। (৫) "a"/"a+"-এ fseek-এর পরেও লেখা যায় শেষে।
মনে রাখার কৌশল "r পড়ে, w মোছে, a জোড়ে — আর + দেয় বাকি অর্ধেক।" "b মানে বাইট।" ফাইল না থাকলে ব্যর্থ হয় কেবল দুটি r-মোড; সব w- ও a-মোড ফাইল তৈরি করে।

সংজ্ঞা প্রতিটি খোলা স্ট্রিমের একটি ফাইল পজিশন ইন্ডিকেটর থাকে — যে বাইট অফসেটে পরের পড়া বা লেখা হবে। ধারাবাহিক (sequential) I/O একে নিজে নিজে সামনে নেয়; র‍্যান্ডম অ্যাক্সেস একে সরাসরি সরায় fseek(fp, offset, whence) দিয়ে, ফলে বাইনারি ফাইলের k-তম রেকর্ড আগের k − 1টি না পড়েই পড়া যায়।
কলফল · রিটার্ন
fseek(fp, off, whence)শুরুবিন্দু + off-এ যায় · সফল হলে 0
ftell(fp)বর্তমান অফসেট, long হিসেবে · এররে −1L
rewind(fp)0-তে ফেরে এবং এরর ফ্ল্যাগ মোছে · void
feof(fp)কোনো পড়া শেষে ইতিমধ্যে ঠেকলে অশূন্য
ferror(fp) / clearerr(fp)এরর ফ্ল্যাগ আছে? / দুটি ফ্ল্যাগই মোছে
fflush(fp)আউটপুট বাফার এখনই লিখে দেয়
remove(n) / rename(o, n)ফাইল মোছে / নাম বদলায় · সফল হলে 0
সমাধানকৃত উদাহরণ ✍ ছাত্রদের রেকর্ডের বাইনারি ফাইল struct student { int roll; char name[16]; float cgpa; }; — 4 + 16 + 4 = 24 বাইট (এখানে প্যাডিং লাগে না; প্যাডিংয়ের নিয়ম স্ট্রাকচার লেসনে)। (ক) ফাইলের আকার: fseek(fp, 0, SEEK_END); long n = ftell(fp); → n = 240 ⇒ 240 ÷ 24 = ১০টি রেকর্ড। (খ) ৫ম রেকর্ড পড়া (k = 4): fseek(fp, 4 * sizeof(struct student), SEEK_SET) → বাইট 96, তারপর fread(&s, sizeof s, 1, fp) ফেরত দেয় 1। (গ) এখন ftell(fp) = 96 + 24 = 120। (ঘ) জায়গাতেই হালনাগাদ: "rb+" দিয়ে খোলো, fseek(fp, -24L, SEEK_CUR), তারপর fwrite(&s, sizeof s, 1, fp)। (ঙ) শেষ রেকর্ড: fseek(fp, -24L, SEEK_END) → বাইট 216।
/* WRONG: the last line prints twice */ while (!feof(fp)) { fgets(line, 80, fp); fputs(line, stdout); } /* RIGHT: test the read itself */ while (fgets(line, sizeof line, fp) != NULL) fputs(line, stdout);
EOF একটি মান, বাইট নয় EOF হলো <stdio.h>-এর একটি ম্যাক্রো, একটি ঋণাত্মক int (সাধারণত −1), যা কিছু না থাকলে ইনপুট ফাংশন ফেরত দেয়। ফাইলে কোনো EOF অক্ষর জমা থাকে না; ফাইল শুধু শেষ হয়। তাই fgetc ফেরত দেয় int: এতে 256টি বাইট-মান এবং EOF — সবই আঁটতে হবে।
পরীক্ষার ফাঁদ (১) feof() সামনে দেখে না: কোনো পড়া ব্যর্থ হওয়ার পরেই এটি সত্য হয়, তাই while (!feof(fp)) লুপ একবার বেশি চলে। (২) ftell দিয়ে বাইটের হিসাবের জন্য বাইনারি মোড লাগে। (৩) লেখা আর পড়ার মাঝে fseek বা fflush ডাকো।
মনে রাখার কৌশল "SET, CUR, END = 0, 1, 2 — শুরু, এখানে, শেষ।" আর "আগে পড়ো, পরে জিজ্ঞেস করো": fgetc/fgets/fread-এর ফেরত মান যাচাই করো; feof() জিজ্ঞেস করো কেবল কেন থামল জানতে।

৫পরিবর্তনশীল দৈর্ঘ্যের আর্গুমেন্ট তালিকা (stdarg.h)

লিখিত প্রশ্নে প্রিয় ★
সংজ্ঞা ভেরিয়াডিক ফাংশন (variadic function) পরিবর্তনশীল সংখ্যক আর্গুমেন্ট নেয়, যা ঘোষণা করা হয় অন্তত একটি নামযুক্ত প্যারামিটারের পরে একটি এলিপসিস ... দিয়ে: int printf(const char *fmt, ...);। <stdarg.h>-এর ম্যাক্রোগুলো নামহীন আর্গুমেন্ট একটা একটা করে পড়ে; কতগুলো এল তা ফাংশনের নিজের জানার কোনো উপায় নেই।
#include <stdarg.h> int sum(int count, ...) { va_list ap; int total = 0; va_start(ap, count); for (int i = 0; i < count; i++) total += va_arg(ap, int); va_end(ap); return total; } sum(4, 10, 20, 30, 40) → 100 sum(2, 5, 15) → 20
ম্যাক্রো / টাইপকাজ
va_list apআর্গুমেন্টগুলোর ওপর কার্সর
va_start(ap, last)শেষ নামযুক্ত প্যারামিটারের পর থেকে শুরু
va_arg(ap, T)পরেরটি T টাইপ হিসেবে আনে
va_copy(d, s)কার্সরের কপি (C99)
va_end(ap)পরিষ্কার করে — প্রতি va_start-এ একটি
পরীক্ষার ফাঁদ নামহীন আর্গুমেন্টে ডিফল্ট প্রোমোশন হয়: char, short → int; float → double। তাই va_arg(ap, float) ও va_arg(ap, char) অনির্ধারিত — double / int চাও। শেষ আর্গুমেন্টের পরে পড়াও অনির্ধারিত।
কলি সংখ্যা জানে কীভাবে একটি count আর্গুমেন্ট (ওপরের sum) · শেষে একটি সেন্টিনেল (NULL, −1) · একটি ফরম্যাট স্ট্রিং — printf % চিহ্ন গোনে। সংখ্যা ভুল হলে va_arg আবর্জনা পড়ে।

৬কমান্ড-লাইন প্যারামিটার (argc, argv)

সবচেয়ে বেশি আসে ★
সংজ্ঞা int main(int argc, char *argv[]) (বা char **argv) কমান্ডের পরে টাইপ করা শব্দগুলো পায়। argc = কতগুলো শব্দ, প্রোগ্রামের নামসহ; argv[0] = প্রোগ্রামের নাম; argv[1] … argv[argc − 1] = আর্গুমেন্টগুলো, সবসময় স্ট্রিং হিসেবে; এবং argv[argc] = NULL।
সমাধানকৃত উদাহরণ ✍ $ ./sum 10 20 30 argc = 4; argv[1] হলো স্ট্রিং "10", তাই যোগফলের জন্য রূপান্তর লাগে: for (i = 1; i < argc; i++) s += atoi(argv[i]); → 60। (খ) $ ./prog "hello world" → argc = 2: শেলের উদ্ধৃতিচিহ্ন একে একটি আর্গুমেন্ট বানায়। (গ) শুধু $ ./prog → argc = 1, argv[1] = NULL: আগে argc < 2 যাচাই করো। (ঘ) atoi("12ab") চুপচাপ 12 দেয় — strtol এরর জানাতে পারে।
পরীক্ষার ফাঁদ (১) বাস্তবে argc কখনো 0 নয় — নামটাও গোনা হয়। (২) argv[1] + argv[2] যোগ করে পয়েন্টার, সংখ্যা নয়। (৩) argc/argv নাম শুধু প্রথা; স্ট্যান্ডার্ড যা ঠিক করে দেয় তা হলো টাইপ।
মনে রাখার কৌশল "argc নামও গোনে; argv শেষ হয় NULL-এ।" stdarg-এর জন্য: "start, arg, arg, …, end" — আর "ছোটরা বড় হয়": char → int, float → double।

৭এরর হ্যান্ডলিং: রিটার্ন মান, errno, perror, assert, exit

হাই-ইল্ড ★
সংজ্ঞা C-তে এক্সেপশন (exception) নেই। লাইব্রেরি ফাংশন ব্যর্থতা জানায় তার রিটার্ন মান দিয়ে এবং অনেকে errno-তে একটি কোড রেখে; প্রোগ্রামকে যাচাই করতে হয়, stderr-এ জানাতে হয় এবং ব্যর্থতার এক্সিট স্ট্যাটাস দিয়ে শেষ করতে হয়। যাচাই বাদ দিলে আপনা-আপনি কিছুই ঘটে না।
কলব্যর্থতার চেহারা
fopen, malloc, calloc, fgetsNULL
scanf, fscanfচাওয়ার চেয়ে কম আইটেম, বা EOF
fgetc, getchar, fclose, fputsEOF
printf, fprintfএকটি ঋণাত্মক মান
fseek / ftellঅশূন্য / −1L
fread, fwriten-এর চেয়ে কম আইটেম

#include <stdio.h> #include <stdlib.h> #include <errno.h> #include <string.h> FILE *fp = fopen("data.txt", "r"); if (fp == NULL) { perror("data.txt"); fprintf(stderr, "code %d: %s\n", errno, strerror(errno)); exit(EXIT_FAILURE); }
errno · perror · strerror errno (<errno.h>) ব্যর্থ কল একটি ধনাত্মক কোডে বসায়, আর লাইব্রেরি একে কখনো 0-তে ফেরায় না। ISO C সংজ্ঞায়িত করে কেবল EDOM, ERANGE, EILSEQ; ENOENT ইত্যাদি POSIX-এর। perror("data.txt") stderr-এ ছাপে data.txt: No such file or directory; strerror(errno) (<string.h>) সেই লেখাটি স্ট্রিং হিসেবে দেয়।
assert, exit, abort — প্রোগ্রাম যেভাবে শেষ হয়

assert(expr): মিথ্যা হলে stderr-এ এক্সপ্রেশন, ফাইল, লাইন ছাপে, তারপর abort()

<assert.h>-এর আগে #define NDEBUG ⇒ সব assert উধাও

exit(status): স্ট্রিম ফ্লাশ ও বন্ধ করে, atexit() হ্যান্ডলার চালায়

main-এ return n; ≡ exit(n) · EXIT_SUCCESS 0 · EXIT_FAILURE ≠ 0

abort(): অস্বাভাবিক সমাপ্তি, atexit হ্যান্ডলার চলে না। শেল স্ট্যাটাস পড়ে $? / %ERRORLEVEL% দিয়ে।

সমাধানকৃত উদাহরণ ✍ শূন্য দিয়ে ভাগ b = 0 হলে পূর্ণসংখ্যার a / b অনির্ধারিত আচরণ — কোনো রিটার্ন কোড নেই, errno নেই, সাধারণত ক্র্যাশ — তাই আগে যাচাই: if (b == 0) { fputs("divide by zero\n", stderr); return EXIT_FAILURE; }। (IEEE 754 float-এ 1.0 / 0.0 = inf।)
পরীক্ষার ফাঁদ (১) errno অর্থবহ কেবল ব্যর্থ কলের ঠিক পরে। (২) assert ইনপুট যাচাই নয় — রিলিজ বিল্ডে উধাও হয়। (৩) এরর জানাও stderr-এ, stdout-এ নয়। (৪) 0 = সাফল্য, অশূন্য = ব্যর্থতা — উল্টো নয়।
মনে রাখার কৌশল "যাচাই, জানাও, বেরোও।" রিটার্ন মান যাচাই করো, stderr-কে জানাও (perror), EXIT_FAILURE দিয়ে বেরোও। "বাগের জন্য assert, ব্যবহারকারীর জন্য if।"

/* util.h */ extern int counter; int add(int a, int b); /* util.c */ int counter = 0; int add(int a, int b) { counter++; return a + b; } /* main.c */ #include "util.h" gcc -c main.c → main.o gcc -c util.c → util.o gcc main.o util.o -o prog gcc prog.c -o prog -lm (libm)
extern ও লিংকেজ extern এমন একটি নাম ঘোষণা করে যার সংজ্ঞা অন্য কোথাও; সংজ্ঞা থাকতে হবে ঠিক একটি ফাইলে। ফাইল স্কোপে static দেয় ইন্টারনাল লিংকেজ: নামটি লিংকার ও অন্য ফাইলের কাছে অদৃশ্য (স্টোরেজ ক্লাস লেসন)। -lm গণিত লাইব্রেরি libm যুক্ত করে; একে সোর্স ফাইলের পরে রাখো।

লিখিত প্রশ্ন একটি C প্রোগ্রাম লেখো যা কমান্ড লাইন থেকে একটি ফাইলের নাম নেয়, ফাইলের পূর্ণসংখ্যাগুলো পড়ে তাদের যোগফল ছাপে এবং প্রতিটি এরর ঠিকভাবে সামলায়। argc/argv এবং তোমার প্রতিটি যাচাই ব্যাখ্যা করো। ১০
  1. আর্গুমেন্ট — int main(int argc, char *argv[]); argc প্রোগ্রামের নামও গোনে, তাই একটি ফাইলের নাম মানে argc == 2, আর argv[1] হলো নামটি। argc != 2 হলে stderr-এ ব্যবহারবিধি ছাপো এবং return EXIT_FAILURE;। ২
  2. খোলা — FILE *fp = fopen(argv[1], "r"); এবং if (fp == NULL) { perror(argv[1]); return EXIT_FAILURE; } — perror errno থেকে কারণটি যোগ করে, stderr-এ। ২
  3. পড়া — long sum = 0; int x; while (fscanf(fp, "%d", &x) == 1) sum += x; — লুপটি রিটার্ন মান যাচাই করে (feof নয়), তাই ফাইল শেষে ও ভুল ডেটায় থামে; তারপর if (ferror(fp)) হলে পড়ার এরর জানাও। ৩
  4. শেষ — fclose(fp); printf("Sum = %ld\n", sum); return EXIT_SUCCESS; — স্ট্যাটাস 0 শেলকে জানায় কাজ সফল। ১
  5. ব্যাখ্যা — argv রাখে স্ট্রিং, argv[argc] হলো NULL; C-তে এক্সেপশন নেই বলে প্রতিটি লাইব্রেরি কল যাচাই করা হয়েছে: fopen-এর NULL, fscanf-এর আইটেম সংখ্যা, ferror-এর ফ্ল্যাগ। ২
